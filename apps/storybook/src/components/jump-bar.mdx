# Jump bar

A jump bar moves through a conversation: up to unread messages, down to new
ones, or back to the present from history.

<Canvas of={Stories.Default} />

## Features

- Three types: unread messages above with "Mark as read", a pill for new
  messages below, and a bar while viewing history.
- Counts cap at 99+ on screen; the accessible name adds the full number.
- `jumpTo` scrolls to a message and focuses it.
- `useAtBottom` says whether the conversation is scrolled to its end.

## Usage

```tsx
import { useState } from "react"

import { JumpBar, jumpTo, useAtBottom } from "@/components/ui/jump-bar"
import { MessageList, type ChatMessage } from "@/components/ui/message-list"

export function Channel({
  messages,
  unreadFrom,
  newBelow,
}: {
  messages: ChatMessage[]
  unreadFrom?: string
  newBelow: number
}) {
  const [scroller, setScroller] = useState<HTMLDivElement | null>(null)
  const atBottom = useAtBottom(scroller)

  function jumpToPresent() {
    const last = scroller?.querySelector<HTMLElement>(
      '[data-slot="message"]:last-of-type'
    )
    if (last) jumpTo(last, { container: scroller, block: "end" })
  }

  return (
    <div className="relative">
      <div ref={setScroller} className="h-96 overflow-y-auto">
        <MessageList messages={messages} unreadFrom={unreadFrom} />
      </div>
      {newBelow > 0 && !atBottom ? (
        <JumpBar type="new-below" count={newBelow} onJump={jumpToPresent} />
      ) : null}
    </div>
  )
}
```

The bars position themselves. Place them in a `relative` wrapper around the
scroll container, not inside the container.

## API reference

JumpBar also takes every prop of a `div`, except `children`.

<ArgTypes of={Stories.Default} />

`jumpTo(target, { container, block })` scrolls `target` into view and
focuses it. `useAtBottom(element, threshold = 150)` returns whether `element`
is scrolled to within `threshold` pixels of its end.
